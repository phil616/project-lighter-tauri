cmake_minimum_required(VERSION 3.21)
set(LIGHTER_VERSION_OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/version.json")
include(cmake/Version.cmake)
project(DreamReflexLighter VERSION "${LIGHTER_VERSION}" LANGUAGES CXX)
configure_file(native/version.h.in generated/lighter/version.h @ONLY)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Concurrent Network)
add_library(lighter_core native/config.cpp native/runner.cpp native/ports.cpp native/ansi.cpp native/batch.cpp)
target_include_directories(lighter_core PUBLIC native)
target_link_libraries(lighter_core PUBLIC Qt6::Core Qt6::Gui Qt6::Concurrent)
if(WIN32)
  target_link_libraries(lighter_core PRIVATE iphlpapi ws2_32)
endif()
add_library(lighter_ui native/window.cpp native/editor.cpp native/theme.cpp
  native/styles/styles.qrc
  native/vendor/qdarkstyle/dark/darkstyle.qrc
  native/vendor/qdarkstyle/light/lightstyle.qrc)
target_link_libraries(lighter_ui PUBLIC lighter_core Qt6::Widgets)
target_include_directories(lighter_ui PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/generated")
qt_add_executable(lighter WIN32 native/main.cpp native/resources.qrc)
target_link_libraries(lighter PRIVATE lighter_ui)
if(WIN32)
  configure_file(native/windows.rc.in windows.rc @ONLY)
  target_sources(lighter PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/windows.rc)
endif()
if(MSVC)
  target_compile_options(lighter_core PUBLIC /utf-8)
  target_compile_options(lighter_ui PUBLIC /utf-8)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  qt_add_executable(lighter_tests tests/test_lighter.cpp)
  target_link_libraries(lighter_tests PRIVATE lighter_ui Qt6::Test Qt6::Network)
  add_test(NAME lighter_tests COMMAND lighter_tests)
  set_tests_properties(lighter_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_test(NAME lighter_version COMMAND lighter --version)
  string(REPLACE "." "\\." _version_pattern "${LIGHTER_VERSION}")
  set_tests_properties(lighter_version PROPERTIES
    ENVIRONMENT "QT_QPA_PLATFORM=offscreen"
    PASS_REGULAR_EXPRESSION "DreamReflexLighter ${_version_pattern}")
  get_filename_component(_make_bin "${CMAKE_MAKE_PROGRAM}" DIRECTORY)
  add_test(NAME tag_version COMMAND "${CMAKE_COMMAND}"
    "-DVERSION_MODULE=${CMAKE_CURRENT_SOURCE_DIR}/cmake/Version.cmake"
    "-DTEST_BINARY_DIR=${CMAKE_CURRENT_BINARY_DIR}"
    "-DTEST_MAKE_BIN=${_make_bin}" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_version.cmake")
endif()
include(GNUInstallDirs)
set(CMAKE_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
install(TARGETS lighter RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
  PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
install(FILES native/vendor/qdarkstyle/LICENSE.rst native/vendor/qdarkstyle/NOTICE.md
  DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/lighter/qdarkstyle)
install(FILES "${LIGHTER_VERSION_OUTPUT}" DESTINATION ${CMAKE_INSTALL_DATADIR}/lighter)
if(WIN32)
  qt_generate_deploy_app_script(TARGET lighter OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
  install(SCRIPT ${deploy_script})
else()
  install(FILES packaging/com.dreamreflex.lighter.desktop DESTINATION share/applications PERMISSIONS OWNER_READ OWNER_WRITE GROUP_READ WORLD_READ)
  install(FILES native/assets/icon.png DESTINATION share/icons/hicolor/128x128/apps RENAME com.dreamreflex.lighter.png PERMISSIONS OWNER_READ OWNER_WRITE GROUP_READ WORLD_READ)
endif()
set(CPACK_PACKAGE_NAME "dream-reflex-lighter")
set(CPACK_PACKAGE_VENDOR "DreamReflex")
set(CPACK_PACKAGE_CONTACT "phil616@users.noreply.github.com")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native Qt project launcher")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_INSTALL_DIRECTORY "DreamReflexLighter")
if(WIN32)
  set(CPACK_GENERATOR "ZIP;NSIS")
  set(CPACK_NSIS_EXECUTABLES_DIRECTORY "bin")
  set(CPACK_PACKAGE_EXECUTABLES "lighter" "项目启动器")
else()
  set(CPACK_GENERATOR "TGZ;DEB")
  set(CPACK_PRE_BUILD_SCRIPTS "${CMAKE_CURRENT_SOURCE_DIR}/packaging/permissions.cmake")
  set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
  set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "lsof, iproute2")
endif()
include(CPack)
